cmake_minimum_required(VERSION 3.21)
project(fetchjet_native LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)

set(SLINT_DIR "E:/tools/slint" CACHE PATH "Unpacked Slint C++ package")
set(FJ_CORE_KIND "releaseShared" CACHE STRING "Kotlin/Native build of fjcore.dll: releaseShared or debugShared")
find_package(Slint REQUIRED PATHS "${SLINT_DIR}/lib/cmake/Slint")

set(FJ_APP_VERSION_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../sharedLogic/src/commonMain/kotlin/com/michelle/raouf/fetch_jet/core/update/AppVersion.kt")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${FJ_APP_VERSION_FILE}")
file(READ "${FJ_APP_VERSION_FILE}" FJ_APP_VERSION_SOURCE)
if (NOT FJ_APP_VERSION_SOURCE MATCHES "APP_VERSION[ \t]*=[ \t]*\"([^\"]+)\"")
    message(FATAL_ERROR "APP_VERSION not found in ${FJ_APP_VERSION_FILE}")
endif ()
set(FJ_APP_VERSION "${CMAKE_MATCH_1}")
string(REGEX REPLACE "-.*$" "" FJ_VERSION_PLAIN "${FJ_APP_VERSION}")
string(REPLACE "." ";" FJ_VERSION_PARTS "${FJ_VERSION_PLAIN}.0.0")
list(GET FJ_VERSION_PARTS 0 FJ_VERSION_MAJOR)
list(GET FJ_VERSION_PARTS 1 FJ_VERSION_MINOR)
list(GET FJ_VERSION_PARTS 2 FJ_VERSION_PATCH)
if (NOT "${FJ_VERSION_MAJOR}.${FJ_VERSION_MINOR}.${FJ_VERSION_PATCH}" MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "APP_VERSION \"${FJ_APP_VERSION}\" is not MAJOR.MINOR.PATCH")
endif ()
string(TIMESTAMP FJ_VERSION_YEAR "%Y")
configure_file(res/version.h.in "${CMAKE_CURRENT_BINARY_DIR}/generated/fj_version.h" @ONLY)

add_executable(fetchjet_native WIN32 host/main.cpp res/app.rc res/app.manifest)
set_target_properties(fetchjet_native PROPERTIES OUTPUT_NAME "FetchJet")
target_include_directories(fetchjet_native PRIVATE host)
target_include_directories(fetchjet_native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(fetchjet_native PRIVATE Slint::Slint dwmapi ole32 shell32 uuid d2d1 dwrite windowscodecs delayimp)
slint_target_sources(fetchjet_native ui/app.slint COMPILATION_UNITS 4)
set_property(TARGET fetchjet_native PROPERTY SLINT_EMBED_RESOURCES embed-files)
if (MSVC)
    target_link_options(fetchjet_native PRIVATE /ENTRY:mainCRTStartup /DELAYLOAD:d2d1.dll /DELAYLOAD:DWrite.dll
            /DELAYLOAD:WindowsCodecs.dll /IGNORE:4199)
    target_compile_options(fetchjet_native PRIVATE /utf-8 /W3)
endif ()
add_custom_command(TARGET fetchjet_native POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SLINT_DIR}/lib/slint_cpp.dll" $<TARGET_FILE_DIR:fetchjet_native>
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/core/build/bin/mingwX64/${FJ_CORE_KIND}/fjcore.dll" $<TARGET_FILE_DIR:fetchjet_native>)
